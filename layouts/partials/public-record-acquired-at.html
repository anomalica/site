{{- /* Return only a canonical, offset-bearing RFC 3339 instant that also parses as a real time. Keep the producer's exact lexical value. */ -}} {{- $value := . -}} {{- $safe := "" -}} {{- if and (eq (printf "%T" $value) "string") (eq (len (findRE `^[0-9]{4}-(?:0[1-9]|1[0-2])-(?:0[1-9]|[12][0-9]|3[01])T(?:[01][0-9]|2[0-3]):[0-5][0-9]:[0-5][0-9](?:\.[0-9]+)?(?:Z|[+-](?:[01][0-9]|2[0-3]):[0-5][0-9])$` $value)) 1) (ne (substr $value 0 4) "0000") -}} {{- $parsed := try (time.AsTime $value) -}} {{- if not $parsed.Err -}}{{- $safe = $value -}}{{- end -}} {{- end -}} {{- return $safe -}}
