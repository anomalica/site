{{- /* The providers a stage's banned watermarking states resolve to. Rendered so a reader can check the derivation rather than trust a list: the states are shown above on the page, and this is what they add up to today. Takes a dict of `stage` and `policy`; returns a slice of names. */ -}} {{- $stage := .stage -}} {{- $policy := .policy -}} {{- $providers := slice -}} {{- range $provider, $entry := $policy.watermarking -}} {{- if in $stage.deny_watermarking $entry.state }}{{ $providers = $providers | append $provider }}{{ end -}} {{- end -}} {{- return $providers -}}
