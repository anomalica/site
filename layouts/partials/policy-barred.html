{{- /* The models a stage refuses, worked out from the data rather than listed. A reader-facing stage bans watermarking STATES (`deny_watermarking`), not provider names: the set of banned providers is computed from the watermarking table every time, so adding a provider to that table with a banned state IS banning it. A typed-out provider list would go stale the first time a provider was added and nobody remembered - and a model can move between states as evidence arrives, which a name list would not follow. `deny` provider globs are still honoured for anything listed that way. Takes a dict of `stage` and `policy`; returns a slice of model entries. */ -}} {{- $stage := .stage -}} {{- $policy := .policy -}} {{- $barred := slice -}} {{- range $policy.models -}} {{- $model := . -}} {{- $state := "" -}} {{- with index $policy.watermarking $model.provider }}{{ $state = .state }}{{ end -}} {{- $refused := false -}} {{- if and $state (in $stage.deny_watermarking $state) }}{{ $refused = true }}{{ end -}} {{- range $stage.deny -}} {{- if eq . (printf "%s/*" $model.provider) }}{{ $refused = true }}{{ end -}} {{- end -}} {{- if $refused }}{{ $barred = $barred | append $model }}{{ end -}} {{- end -}} {{- return $barred -}}
