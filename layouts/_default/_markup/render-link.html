{{- /* Markdown link render hook. Three behaviours: 1. Internal entity-section links (/people/, /topics/, ...) are existence-gated at render time: if the target page is in the build it renders a language-relative ; if not, the link text is kept as plain text with no href - so a prose entity mention can never dead-link, and it lights up on its own once that page is assembled (never stale). 2. Other internal links (/about/, /architecture/, section indexes, ...) are language-prefixed via relLangURL, no existence gate. 3. External links, workbench deep-links (http...), anchors and mailto pass through untouched. */ -}} {{- $dest := .Destination -}} {{- $sections := slice "people" "organisations" "projects" "events" "places" "documents" "objects" "topics" "patterns" "concepts" "records" -}} {{- $internal := and (strings.HasPrefix $dest "/") (not (strings.HasPrefix $dest "//")) -}} {{- if $internal -}} {{- $seg := index (split (strings.TrimPrefix "/" $dest) "/") 0 -}} {{- if in $sections $seg -}} {{- $lookup := index (split $dest "#") 0 -}} {{- $lookup = index (split $lookup "?") 0 -}} {{- if site.GetPage $lookup -}} {{ .Text | safeHTML }} {{- else -}} {{ .Text | safeHTML }} {{- end -}} {{- else -}} {{ .Text | safeHTML }} {{- end -}} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
